# Authoritative source of truth for the kubetix-api service. Installed by
# `kubetix-api/Dockerfile` in production. Exact-pinned on purpose so the API
# image is reproducible across rebuilds.
#
# The repository root also has a `requirements.txt` that overlaps with this
# file. It exists for two reasons: (1) the root Dockerfile builds a small CLI
# image that runs `kc-share.py`, and (2) the repo-root test runner needs the
# API's Python deps on PYTHONPATH without having to bootstrap the
# kubetix-api image first. Whenever a shared dependency is bumped here, the
# corresponding loose range in `../requirements.txt` must be loosened to
# include the new exact version. The reverse direction does NOT need to be
# mirrored: this file is the single source of truth for the service.

uvicorn[standard]==0.52.1
sqlalchemy==2.0.52
python-jose[cryptography]==3.5.0
bcrypt>=5.0.0,<5.1.0  # pinned to 4.x; bcrypt 5.x breaks passlib's CryptContext
python-multipart==0.0.32
pydantic==2.13.4
cryptography==50.0.0
httpx==0.28.1
slowapi==0.1.10
# PostgreSQL drivers (issue #310). ``psycopg2-binary`` is the default
# SQLAlchemy driver for ``postgresql://`` / ``postgresql+psycopg2://``
# URLs and is what ``DATABASE_URL`` in the bundled Helm chart resolves
# to. ``psycopg[binary]`` (psycopg 3) is installed alongside it for
# ``postgresql+psycopg://`` URLs and because it ships manylinux_2_28
# wheels for newer CPython releases (3.13+), avoiding source builds
# on slim container images where libpq headers are absent. The source
# build fallback for ``psycopg2-binary`` is enabled in the Dockerfile
# (libpq-dev) for the (transient) window before upstream publishes a
# wheel for the newest CPython.
psycopg2-binary>=2.9.10
psycopg[binary]>=3.2.0,<4.0.0

# release line.
alembic>=1.13.0,<2.0.0
